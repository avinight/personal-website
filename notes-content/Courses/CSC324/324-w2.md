```Haskell
maxSubarray :: [Int] -> Int
maxSubarray xs = subarraySum xs 0 0
  where
    subarraySum [] _ currMax = currMax
    subarraySum (k:ks) acc currMax =
      let newAcc = acc + k
      in if newAcc < 0
           then subarraySum ks 0 currMax
           else subarraySum ks newAcc (max currMax newAcc)
           
maxSubarray :: [Int] -> Int
maxSubarray = snd . foldl step (0, minBound :: Int)
  where
    step (acc, currMax) x =
      let newAcc = max x (acc + x)
          newMax = max currMax newAcc
      in (newAcc, newMax)
```

```Haskell
takePositive :: Int -> [Int] -> [Int]
takePositive 0 _ = []
takePositive _ [] = []
takePositive n (x:xs)
    | x > 0     = x : takePositive (n-1) xs
    | otherwise = takePositive n xs
    
-- Tail Optimized
takePositive :: Int -> [Int] -> [Int]
takePositive n xs = takePositiveHelper n xs []
	where takePositiveHelper :: Int -> [Int] -> [Int] -> [Int]
	takePositiveHelper 0 _ acc = acc -- collected enough positives
	takePositiveHelper _ [] acc = acc -- ran out of elements
	takePositiveHelper n (x:xs) acc
		| x > 0 = takePositiveHelper (n-1) xs (append acc x)
		| otherwise = takePositiveHelper n xs acc
		
	-- append a single element to the end of a list
    append :: [Int] -> Int -> [Int]
    append [] y = [y]
    append (z:zs) y = z : append zs y
```

```haskell
oneStep :: Term -> StepResult
oneStep (TmIf TmTrue t2 _) = StepOk t2 -- E-IfTrue
oneStep (TmIf TmFalse _ t3) = StepOk t3 -- E-IfFalse

  | isVal t = StepError "Argument of TmSucc is already a value"

        StepOk t' -> StepOk (TmSucc t') -- E-Succ
```

```haskell
bigStep :: Term -> EvalResult
bigStep t -- B-Value
	| isVal t = EvalOk t
bigStep (TmIf t1 t2 t3) -- B-IfTrue, B-IfFalse
	| isBoolVal t1 = if isTmTrue t1 then EvalOk t2 else EvalOk t3
	| otherwise = 
		case bigStep t1 of
			EvalOk t1' -> bigStep (TmIf t1' t2 t3)
			EvalError msg -> EvalError msg
	where 
		isTmTrue :: Term -> Bool
		isTmTrue TmTrue = True
		isTmTrue _ = False

	case bigStep t of
		EvalOk t' | isNumVal t' -> EvalOk (TmSucc t')
		EvalOk _ -> EvalError "Argument of TmSucc is not a number"
```
