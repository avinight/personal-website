mapTree applys f to every leaf and internal node of the binary tree.
```Haskell
data BinaryTree a = Leaf a
                  | Branch (BinaryTree a) a (BinaryTree a)
                  deriving (Eq, Show)

mapTree :: (a -> b) -> BinaryTree a -> BinaryTree b
mapTree f (Leaf l) = Leaf (f l)
mapTree f (Branch left val right) = Branch (mapTree f left) (f val) (mapTree f right)
```


Transform x to a list type if its a leaf, if its a branch, we want to cons it 

```Haskell
transformTree :: (a -> b) -> (b -> a -> b -> b) -> BinaryTree a -> b

transformTree f _ (Leaf l) = f l
transformTree f g (Branch left val right) = 
    g (transformTree f g left) val (transformTree f g right)
    
preorder :: BinaryTree a -> [a]
preorder = transformTree (\x -> [x]) (\left x right -> x : (left ++ right))

inOrder :: BinaryTree a -> [a]
inOrder = transformTree (\x -> [x]) (\left x right -> left ++ (x : right))

postOrder :: BinaryTree a -> [a]
postOrder = transformTree (\x -> [x]) (\left x right -> left ++ right ++ [x])
```

Part 2 Evaluator

The function takes in a NamingContext and a NamedTerm. As output, it outputs a Term that is
the nameless representation of the input NamedTerm. We do not test the situation where a variable
cannot be found in the naming context

for every variable in the named term, lookup its corresponding variable in the naming context. If there is a lambda, then we need to shift.


```Haskell
removeNames :: NamingContext -> NamedTerm -> Term

removeNames ctx (NTVar t) = TmVar $ indexOf t ctx
-- extend context for abstractions
removeNames ctx (NTAbs iden body) = TmAbs $ removeNames (iden : ctx) body
removeNames ctx (NTApp t1 t2) = TmApp (removeNames ctx t1) (removeNames ctx t2)

-- Other Cases:
removeNames _   NTTrue          = TmTrue
removeNames _   NTFalse         = TmFalse
removeNames ctx (NTIf t1 t2 t3) = TmIf (removeNames ctx t1) (removeNames ctx t2) (removeNames ctx t3)
removeNames _   NTZero          = TmZero
removeNames ctx (NTSucc t)      = TmSucc (removeNames ctx t)
removeNames ctx (NTPred t)      = TmPred (removeNames ctx t)
removeNames ctx (NTIsZero t)    = TmIsZero (removeNames ctx t) 
```


```Haskell
shifting :: Int -> Int -> Term -> Term
shifting d c (TmVar k)
	| k < c = k
	| otherwise = k + d

shifting d c (TmAbs t) = TmAbs $ shifting d (c + 1) t
shifting d c (TmApp t1 t2) = TmApp (shifting d c t1) (shifting d c t2)

```

```Haskell
subst :: Int -> Term -> Term -> Term
subst j s (TmVar k)
	| k == j = s
	| otherwise = k
	
subst j s (TmAbs t) = TmAbs $ subst (j + 1) (shifting 1 0 s) t
subst j s (TmApp t1 t2) = TmApp (subst j s t1) (subst j s t2)

```

oneStep (TmApp (TmAbs t12) v2)                           -- E-AppAbs
  | isVal v2 = StepOk $ shifting (-1) 0 $ subst 0 (shifting 1 0 v2) t12

oneStep (TmApp v1 t2)                                    -- E-App2
  | isVal v1 = case oneStep t2 of
      StepOk t2' -> StepOk (TmApp v1 t2')
      StepError msg -> StepError ("The argument cannot be stepped: " 
                                  ++ show t2 ++ "\n" 
                                  ++ msg)

oneStep (TmApp t1 t2) =                                  -- E-App1
  case oneStep t1 of
    StepOk t1' -> StepOk (TmApp t1' t2)
    StepError msg -> StepError ("The function cannot be stepped: " 
                                ++ show t1 ++ "\n" 
                                ++ msg)

```Haskell
lookupEnv :: Env -> String -> Maybe Value
lookupEnv [] _ = Nothing
lookupEnv ((y, v):env) x
  | x == y    = Just v
  | otherwise = lookupEnv env x


bigStep :: Env -> NamedTerm -> EvalResult

bigStep env (NTAbs x t) = EvalOk (Closure env x t) -- B-Abs

-- B-Var
bigStep env (NTVar x) =
  case lookupEnv env x of
    Just v  -> EvalOk v                               
    Nothing -> EvalError ("Unbound variable: " ++ x) 

-- B-App
bigStep env (NTApp t1 t2) = 
  case bigStep env t1 of
    EvalOk (Closure env' x body) ->
      case bigStep env t2 of
        EvalOk v2 -> bigStep ((x, v2) : env') body
        EvalError msg -> EvalError msg
    EvalOk _ -> EvalError "Trying to apply a non-function"
    EvalError msg -> EvalError msg
```
